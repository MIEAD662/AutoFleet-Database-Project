--List the robots associated with each manufacturer, and also show the manufacturers that do not have any robots.
SELECT m.Manufacturer__Name,
       m.Country,
       r.Robot_Name,
       r.Robot_Type
FROM ROBOT r
RIGHT JOIN MANUFACTURER m
ON r.Manfacturer_Id = m.Manufacturer_Id;



--How many drones are there of each drone type?
SELECT Robot_Status,
       COUNT(*) AS Total_Robots
FROM ROBOT
GROUP BY Robot_Status;


--How many times has each type of maintenance problem occurred?
SELECT Problem,
       COUNT(*) AS Total_Problems
FROM MAINTENANCE
GROUP BY Problem;


--What is the average maintenance cost per robot?

SELECT r.Robot_Name,
       AVG(m.Cost) AS Average_Maintenance_Cost
FROM ROBOT r
INNER JOIN MAINTENANCE m
ON TO_NUMBER(SUBSTR(r.Robot_Id, 2)) = m.Robot_Id
GROUP BY r.Robot_Name;


--Created a view to display information about robots and their manufacturers.
CREATE VIEW Robot_Manufacturer_View AS
SELECT r.Robot_Name,
       r.Robot_Type,
       r.Robot_Status,
       m.Manufacturer__Name,
       m.Country
FROM ROBOT r
JOIN MANUFACTURER m
ON r.Manfacturer_Id = m.Manufacturer_Id;


--Creating a view to show how many robots each manufacturer has produced.
CREATE VIEW Manufacturer_Robot_Count AS
SELECT m.Manufacturer__Name,
       COUNT(r.Robot_Id) AS Total_Robots
FROM MANUFACTURER m
LEFT JOIN ROBOT r
ON m.Manufacturer_Id = r.Manfacturer_Id
GROUP BY m.Manufacturer__Name;


--Status check of the R002 robot on the ROBOT table.Active হয় তাহলে "Robot is Ready" দেখাবে, না হলে "Robot is Not Ready"

DECLARE
    r_status ROBOT.Robot_Status%TYPE;
BEGIN
    SELECT Robot_Status
    INTO r_status
    FROM ROBOT
    WHERE Robot_Id = 'R002';

    IF r_status = 'Active' THEN
        DBMS_OUTPUT.PUT_LINE('Robot is Ready');
    ELSE
        DBMS_OUTPUT.PUT_LINE('Robot is Not Ready');
    END IF;
END;
/


--OBOT table-এর R002 robot-এর status অনুযায়ী message;Active → Robot is Operational,Robot is Under Development,Robot is Retired,Unknown Status

DECLARE
    r_status ROBOT.Robot_Status%TYPE;
BEGIN
    SELECT Robot_Status
    INTO r_status
    FROM ROBOT
    WHERE Robot_Id = 'R002';

    IF r_status = 'Active' THEN
        DBMS_OUTPUT.PUT_LINE('Robot is Operational');

    ELSIF r_status = 'Development' THEN
        DBMS_OUTPUT.PUT_LINE('Robot is Under Development');

    ELSIF r_status = 'Retired' THEN
        DBMS_OUTPUT.PUT_LINE('Robot is Retired');

    ELSE
        DBMS_OUTPUT.PUT_LINE('Unknown Status');
    END IF;
END;
/


--The cost for maintenance ID 6 from the MAINTENANCE table.High Maintenance Cost,Medium Maintenance Cost,Low Maintenance Cost

DECLARE
    m_cost MAINTENANCE.Cost%TYPE;
BEGIN
    SELECT Cost
    INTO m_cost
    FROM MAINTENANCE
    WHERE Maintenance_Id = 6;

    IF m_cost > 800 THEN
        DBMS_OUTPUT.PUT_LINE('High Maintenance Cost');

    ELSIF m_cost >= 500 THEN
        DBMS_OUTPUT.PUT_LINE('Medium Maintenance Cost');

    ELSE
        DBMS_OUTPUT.PUT_LINE('Low Maintenance Cost');
    END IF;
END;
/
